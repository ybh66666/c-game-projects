void sheng(int a[][4])
{
	while (1)
	{
		int flag = rand() % 10;
		int num1 = rand() % 4;
		int num2 = rand() % 4;
		if (a[num1][num2] == 0)
		{
			if (flag % 10 == 0)
			{
				
				a[num1][num2] = 4;
				
				break;
			}
			else
			{
				a[num1][num2] = 2;
				
				break;
			}
		}
	}


}
void zuo(int a[][4])
{
	for (int i = 0;i < 4;i++)
	{
		int biao1 = 0;
		int arr[4] = { 0 };
		for (int j = 0;j < 4;j++)
		{
			if (a[i][j] != 0)
			{
				arr[biao1] = a[i][j];
				biao1++;
			}
		}
		for (int j = 0;j < 3;j++)
		{
			if (arr[j] == arr[j + 1] && arr[j] != 0)
			{
				arr[j] *= 2;
				arr[j + 1] = 0;
				j++;
			}
		}
		int biao2 = 0;
		int arr2[4] = { 0 };
		for (int j = 0;j < 4;j++)
		{
			if (arr[j] != 0)
			{
				arr2[biao2] = arr[j];
				biao2++;

			}
		}
		for (int j = 0;j < 4;j++)
		{
			a[i][j] = arr2[j];
		}
	}
}
